TITLE: Algebra of Functions
TAGS: Functions

EXAMPLE:
\\begin{problem}For one pseudo-generated instance $f(x)=2x+(3)$ and $g(x)=1x+(3)$, the source-style algebra-of-functions questions are obtained by forming $f+g$, $f-g$, $fg$, and $f/g$.\\end{problem}

CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def algebraOfFunctions():
    funcvec = [x, x^2, x^3]
    choice1 = RandInt(0,2)
    choice2 = RandInt(0,2)
    c1 = NonZeroInt(-5,5)
    c2 = RandInt(-5,5)
    c3 = NonZeroInt(-5,5)
    c4 = NonZeroInt(-5,5,[c2])
    c5 = NonZeroInt(-5,5)
    c6 = NonZeroInt(-5,5)
    c7 = NonZeroInt(-5,5)
    c8 = NonZeroInt(-5,5)
    c9 = NonZeroInt(-5,5)
    c10 = NonZeroInt(-5,5)
    f1 = c1 * funcvec[choice1](x=(x + c2))
    f2 = c3 * funcvec[choice2](x=(x + c4))
    ans1 = f1 + f2
    ans2 = f1 - f2
    ans3 = f1 *f2
    ans4 = f1/f2
    #### Start of problem p2
    return [f1, f2, ans1, ans2, ans3, ans4]

\begin{problem}
Consider the functions $f(x) = \sage{p1f1}$ and $g(x) = \sage{p1f2}$. Compute the following:
\begin{itemize}
\item $\left(f+g\right)(x) = \answer{\sage{p1ans1}}$
\item $\left(f-g\right)(x) = \answer{\sage{p1ans2}}$
\item $\left(fg\right)(x) = \answer{\sage{p1ans3}}$
\item $\left(\frac{f}{g}\right)(x) = \answer{\sage{p1ans4}}$
\end{itemize}
\end{problem}
