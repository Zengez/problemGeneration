def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def functionNotation():
    c1 = NonZeroInt(-10,10)
    c2 = NonZeroInt(-10,10,[c1])
    domvec = [ 'x' + LatexExpr(r'\leq') + str(c1), str(x < c1), 'x' + LatexExpr(r'\geq') + str(c1), str(x > c1) ]
    codomvec = [ 'x' + LatexExpr(r'\leq') + str(c2), str(x < c2), 'x' + LatexExpr(r'\geq') + str(c2), str(x > c2) ]
    dompick =RandInt(0,3)
    codompick = NonZeroInt(0,3,[dompick])
    var1 = domvec[dompick]
    var2 = codomvec[codompick]
    ans1 = domvec[dompick]
    ans2 = codomvec[codompick]
    ans3 = domvec[3 - dompick]
    ans4 = codomvec[3 - codompick]

    return [var1, var2, ans1, ans2, ans3, ans4]
