TITLE: Fundamental Theorem of Algebra (easy)
TAGS: Polynomials, Functions, easy

CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def fTAEasy():
    c1 = RandInt(1,10)
    c2 = RandInt(1,10)
    c3 = RandInt(1,10)
    c4 = RandInt(1,10)

    pwr1 = RandInt(1,10)
    pwr2 = RandInt(1,10)
    pwr3 = RandInt(1,10)
    pwr4 = RandInt(1,10)

    f1 = c1*x^pwr1
    f2 = c2*x^pwr2
    f3 = c3*x^pwr3
    f4 = c4*x^pwr4

    fDisp = f1+f2+f3+f4

    ans1 = max(pwr1,pwr2,pwr3,pwr4)

    # Followup for p1
    ans2 = 0
    ans3 = 0
    if pwr1 == ans1:
        ans2 = ans2 + f1
        ans3 = ans3 + c1
    if pwr2 == ans1:
        ans2 = ans2 + f2
        ans3 = ans3 + c2
    if pwr3 == ans1:
        ans2 = ans2 + f3
        ans3 = ans3 + c3
    if pwr4 == ans1:
        ans2 = ans2 + f4
        ans3 = ans3 + c4
    return [fDisp, ans1]
