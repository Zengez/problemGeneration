TITLE: Factoring by Completing the Square (easy)
TAGS: Polynomials, Factoring, Functions, easy

EXAMPLE:


CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def factoringCompletingTheSquareEasy():
    #Easy mode, assume monic quadratic
    c1 = RandInt(-10,10)
    c3 = RandInt(-10,10)
    f1 = (x-c1)^2 + c3
    f2 = expand(f1)
    return [f2, c1, c3]
