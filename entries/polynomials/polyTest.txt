def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def polyTest():
    var('x')
    funcvec = [x, x^2, x^3, x^4, x^5, sqrt(x),e^x, 1/x, 1/x^2, log(x)]
    pick1 = RandInt(0,9)
    pick2 = NonZeroInt(0,9,[pick1])
    pick3 = NonZeroInt(5,9,[pick1,pick2])
    c1 = NonZeroInt(-5,5)
    c2 = NonZeroInt(-5,5)
    c3 = NonZeroInt(-5,5)
    c4 = RandInt(-10,10)
    c5 = RandInt(-10,10)
    c6 = RandInt(-10,10)
    ans1 = 0
    ans1t = 0
    ans2t = 0
    ans3t = 0
    f1t = funcvec[pick1]
    f2t = funcvec[pick2]
    f3t = funcvec[pick3]
    f1 = c1*f1t + c4
    f2 = c2*f2t + c5
    f3 = c3*f3t + c6
    f4 = f1 + f2 + f3
    if pick1 > 4:
        ans1t = 1
    if pick2 > 4:
        ans2t = 1
    if pick3 > 4:
        ans3t = 1
    ans1 = ans1t + ans2t + ans3t
    return [f4, ans1]
