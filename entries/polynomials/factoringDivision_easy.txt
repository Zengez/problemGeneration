TITLE: Factoring using Polynomial Division (easy)
TAGS: Polynomials, Factoring, Functions, easy

EXAMPLE:


CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def factoringDivisionEasy():
    c1 = NonZeroInt(-5,5)
    c2 = RandInt(-10,10)
    pwr1 = 1
    pwr2 = RandInt(2,5)
    f0 = sum([RandInt(-3,3)*x^t for t in [0..pwr1-1]])
    f1 = x^pwr1 + sum([RandInt(-3,3)*x^t for t in [0..pwr1-1]])
    f2 = x^pwr2 + sum([RandInt(-3,3)*x^t for t in [0..pwr2-1]])
    f3 = expand(f1*f2) + f0
    return [f3, f1, f2, f0]
