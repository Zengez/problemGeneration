def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def horizontalAsymptotesRatFuncsHard():
    tog=RandInt(0,2);TextVec=[LatexExpr(r'\text{no horizontal asymptote}'),LatexExpr(r'\text{a horizontal asymptote of 0}'),LatexExpr(r'\text{a horizontal asymptote of -3}')];descrText=TextVec[tog];pwr1=RandInt(3,20);fTop=NonZeroInt(-10,10)*x^pwr1+sum([RandInt(-9,9)*x^(RandInt(0,pwr1-1)) for t in [0..RandInt(3,6)]]);deg1=RandInt(1,pwr1-1);deg2=RandInt(pwr1+1,30);deg3=fTop.degree(x);degVec=[deg1,deg2,deg3];ansVec=[deg1,deg2,deg3];shuffle(ansVec);ans=ansVec.index(degVec[tog])+1
    return [fTop, descrText, ansVec, ans]
