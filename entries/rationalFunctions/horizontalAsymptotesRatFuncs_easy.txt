TITLE: Horizontal Asymptotes of Rational Functions (easy)
TAGS: Rational Functions, Asymptotes, Horizontal Asymptotes, easy

CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def horizontalAsymptotesRatFuncsEasy():
    pwr1=RandInt(3,5);tog=RandInt(0,2)
    if tog==0:
        fTop=NonZeroInt(-10,10)*x^pwr1+sum([RandInt(-9,9)*x^(RandInt(0,pwr1-1)) for t in [0..RandInt(1,2)]]);fBot=NonZeroInt(-10,10)*x^pwr1+sum([RandInt(-9,9)*x^(RandInt(0,pwr1-1)) for t in [0..RandInt(1,2)]]);ans=(fTop.leading_coefficient(x))/(fBot.leading_coefficient(x))
    else:
        pwr2=RandInt(3,5);fTop=NonZeroInt(-10,10)*x^pwr1+sum([RandInt(-9,9)*x^(RandInt(0,pwr1-1)) for t in [0..RandInt(1,2)]]);fBot=NonZeroInt(-10,10)*x^pwr2+sum([RandInt(-9,9)*x^(RandInt(0,pwr2-1)) for t in [0..RandInt(1,2)]])
        if pwr1>pwr2: ans=LatexExpr(r'DNE')
        elif pwr1<pwr2: ans=0
        else: ans=(fTop.leading_coefficient(x))/(fBot.leading_coefficient(x))
    return [fTop, fBot, ans]
