TITLE: Domains of Rational Functions (hard)
TAGS: Rational Functions, Domain, hard

EXAMPLE:


CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def domainsOfRationalFunctionsHard():
    c1=NonZeroInt(-3,3);c2=NonZeroInt(-3,3);c3=NonZeroInt(-3,3);c4=RandInt(-3,3);c5=RandInt(-3,3);c6=RandInt(-3,3);fTop=x^6+sum([RandInt(-3,3)*x^t for t in [0..3]]);fBot=expand((c1*x-c2)*(c3*x-c4)*(c5*x-c6));f1=fTop/fBot;ans=c2/c1+c4/c3
    if c5!=0: ans=ans+c6/c5
    return [f1, ans]
