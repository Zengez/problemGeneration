def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def piecewiseAnalyticViewpointMedium():
    funcvec=[x,x^2,x^3,sqrt(abs(x)),ln(abs(x)+1),e^x]
    f1pick=RandInt(0,5);f2pick=RandInt(0,5);f3pick=RandInt(0,5);f1=NonZeroInt(-5,5)*funcvec[f1pick](x=(x-RandInt(-5,5)))+RandInt(-5,5);f2=NonZeroInt(-5,5)*funcvec[f2pick](x=(x-RandInt(-5,5)))+RandInt(-5,5);f3=NonZeroInt(-5,5)*funcvec[f3pick](x=(x-RandInt(-5,5)))+RandInt(-5,5);l1=RandInt(-15,15);r1=l1+RandInt(1,7);l2=RandInt(l1+1,r1+5);r2=l2+RandInt(1,7);l3=RandInt(l2+1,r2+5);r3=l3+RandInt(1,7);ans=1
    if l2<r1:ans=0
    if l3<r2:ans=0
    return [f1, l1, r1, f2, l2, r2, f3, l3, r3, ans]
