    """ Returns a random integer in [`b`,`c`] which is not in `av`.

    """
    while True:
        a = RandInt(b,c)

            return a

def simplifyNumericExponentialsHard():
    ans1=99999/5
    while ((numerator(ans1)>2000) or (denominator(ans1)>2000) or GCD(b1T,b1B)>1): b1T=RandInt(1,8);b1B=RandInt(2,6);b1=(-1)^(RandInt(0,1))*(b1T/b1B);pwr1=NonZeroInt(-6,6,[0,1]);ans1=(b1)^pwr1
    ans2=99999/5
    while ((numerator(ans2)>2000) or (denominator(ans2)>2000) or GCD(b2T,b2B)>1): b2T=RandInt(1,8);b2B=RandInt(2,6);b2=(-1)^(RandInt(0,1))*(b2T/b2B);pwr2=-sign(pwr1)*RandInt(2,6);ans2=(b2)^pwr2
