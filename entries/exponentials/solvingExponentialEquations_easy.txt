TITLE: Solving Exponential Equations (easy)
TAGS: Exponential Functions, Solving Equations, easy

EXAMPLE:
\\begin{problem}Solve $(7)^{4x+(5)}=(7)^{-3x+(-3)}$. One generated solution is $x=\frac{-8}{7}$.\\end{problem}

CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def solvingExponentialEquationsEasy():
    b1=RandInt(2,7);c1=NonZeroInt(-5,5);c2=RandInt(-5,5);c3=NonZeroInt(-5,5,[c1]);c4=RandInt(-5,5);f1=c1*x+c2;f2=c3*x+c4;ans1=(c4-c2)/(c1-c3)
    return [b1, f1, f2, ans1]
