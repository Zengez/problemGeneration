TITLE: Expanding Exponentials (easy)
TAGS: Exponential Functions, Properties of Exponents, Expanding, easy

EXAMPLE:

CODE-SAGE:
var('x,y')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def expandExponentialsEasy():
    b1 = RandInt(2,6)
    ans2b = 9999
    while ans2b > 500:
        b2 = RandInt(2,5)
        ans2b = b1^b2
    ans3b = 9999
    while ans3b > 500:
        b3 = NonZeroInt(2,5,[b2])
        ans3b = b1^b3
    ans1 = 9999
    while ( (ans1 > 1000) or (ans1^(-1) > 1000) or (ans1==ans2b) or (ans1==ans3b) ):
        c1 = NonZeroInt(-5,5)
        ans1 = b1^c1
    c2 = RandInt(1,11)
    c3 = RandInt(1,11)
    f1 = b1^(c1 + b2*x^c2 + b3*y^c3)
    ans2x = x^c2
    ans3y = y^c3
    return [f1,ans1,ans2b,ans2x,ans3b,ans3y]

p1gen = expandExponentialsEasy()
p1f1 = p1gen[0]
p1ans1 = p1gen[1]

p1ans3b = p1gen[4]
p1ans3y = p1gen[5]

\begin{problem}
    \begin{hint}
        Split the exponent (correctly) to write this as a constant term, a term with x in the exponent and a term with y in the exponent.
    \end{hint}
    \begin{hint}
        To correctly split the exponent, use the property $B^{x + y} = B^xB^y$
    \end{hint}
    Expand the following exponential so that each exponent has at most one term.
    \[
        \sage{p1f1}
            = \answer{\sage{p1ans1}} \cdot {\sage{p1ans2b}}^{\answer{\sage{p1ans2x}}} \cdot {\sage{p1ans3b}}^{\answer{\sage{p1ans3y}}}
    \]
\end{problem}
