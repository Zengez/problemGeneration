def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def propertiesOfLogsHard():
    Ac1=RandInt(-10,10); Ac2=NonZeroInt(-10,10,[Ac1]); Ac3=NonZeroInt(-10,10,[Ac1,Ac2]); Ac4=NonZeroInt(-10,10,[Ac1,Ac2,Ac3])
    ABase=RandInt(2,8); ABaseMult=RandInt(2,5); ABase2=ABase^ABaseMult
    while ABase2>260:
        ABaseMult=RandInt(2,5); ABase2=ABase^ABaseMult
    Apwr1=RandInt(1,6); Apwr2=RandInt(1,6); Apwr3=RandInt(1,6); Apwr4=RandInt(1,6)
    Af1=(x-Ac1)^Apwr1; Af2=(y-Ac2)^Apwr2; Af3=(x-Ac3)^Apwr3; Af4=(y-Ac4)^Apwr4
    Aint=(Af1*Af2)/(Af3*Af4)
    Aans1x=Apwr1/ABaseMult; Aans1y=Apwr2/ABaseMult; Aans3x=-Apwr3/ABaseMult; Aans3y=-Apwr4/ABaseMult
    Bbase=RandInt(2,10); Bbase2mult=RandInt(2,4); Bbase2=Bbase^Bbase2mult
    while Bbase2 > 145:
        Bbase=RandInt(2,10); Bbase2mult=RandInt(2,4); Bbase2=Bbase^Bbase2mult
    Bc1=RandInt(-10,10); Bc2=NonZeroInt(-10,10,[Bc1]); Bc3=NonZeroInt(-10,10,[Bc1,Bc2]); Bc4=NonZeroInt(-10,10,[Bc1,Bc2,Bc3])
    Bpwr1=NonZeroInt(-6,6); Bpwr2=NonZeroInt(-6,6); Bpwr3=NonZeroInt(-6,6); Bpwr4=NonZeroInt(-6,6)
    Bf1=(x-Bc1); Bf2=(y-Bc2); Bf3=(x-Bc3); Bf4=(y-Bc4)
    Bans=Bf1^Bpwr1*Bf2^Bpwr2*Bf3^(Bpwr3/Bbase2mult)*Bf4^(Bpwr4/Bbase2mult)
    return [Ac1,Ac2,Ac3,Ac4,ABase,ABaseMult,ABase2,Apwr1,Apwr2,Apwr3,Apwr4,Af1,Af2,Af3,Af4,Aint,Aans1x,Aans1y,Aans3x,Aans3y,Bbase,Bbase2mult,Bbase2,Bc1,Bc2,Bc3,Bc4,Bpwr1,Bpwr2,Bpwr3,Bpwr4,Bf1,Bf2,Bf3,Bf4,Bans]

\begin{problem}
    \begin{hint}
        Use the properties that let you separate logarithms to split each factor into its own logarithm.
    \end{hint}
    \begin{hint}
        Use the property $\Log(A^B) = B\Log(A)$ to pull down any powers of factors and make them into coefficients.
    \end{hint}
    Expand the following logarithmic expression.\\
    \[
        \Log_{\sage{p1ABase2}} \left( \sage{p1Aint} \right) =
        \answer{\sage{p1Aans1x}}\Log_{\sage{p1ABase}}\left(\sage{x - p1Ac1}\right) +
        \answer{\sage{p1Aans1y}}\Log_{\sage{p1ABase}}\left(\sage{y - p1Ac2}\right) +
        \answer{\sage{p1Aans3x}}\Log_{\sage{p1ABase}}\left(\sage{x - p1Ac3}\right) +
        \answer{\sage{p1Aans3y}}\Log_{\sage{p1ABase}}\left(\sage{y - p1Ac4}\right)
    \]
\end{problem}

\begin{problem}
    \begin{hint}
        Use a change of base formula to convert all bases to the same value. Note that you will likely need to explicitly compute the log in the denominator and then rewrite it as a coefficient.
    \end{hint}
    \begin{hint}
        Use the property $\Log(A^B) = B\Log(A)$ to transform any coefficients of logs into a power of the argument.
    \end{hint}
    \begin{hint}
        Use the property $\Log(A) + \Log(B) = \Log(AB)$ and/or $\Log(A) - \Log(B) = \Log\left(\frac{A}{B}\right)$ to merge separate logarithms into one logarithm.
    \end{hint}
    Condense the following logarithmic expression down to one log term.
    \[
        (\sage{p1Bpwr1})\Log_{\sage{p1Bbase}}\left(\sage{p1Bf1}\right) +
        (\sage{p1Bpwr2})\Log_{\sage{p1Bbase}}\left(\sage{p1Bf2}\right) +
        (\sage{p1Bpwr3})\Log_{\sage{p1Bbase2}}\left(\sage{p1Bf3}\right) +
        (\sage{p1Bpwr4})\Log_{\sage{p1Bbase2}}\left(\sage{p1Bf4}\right)
        = \Log_{\sage{p1Bbase}}\left(\answer{\sage{p1Bans}}\right)
    \]
\end{problem}
