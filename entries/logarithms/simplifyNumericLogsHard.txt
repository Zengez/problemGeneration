TITLE: Simplifying Numeric Logs (hard)
TAGS: Simplifying, Logarithmic Functions, Arithmetic, hard


EXAMPLE:


CODE-SAGE:
def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def simplifyNumericLogsHard():
    b1 = RandInt(2,7)
    arg1 = 9999
    b2 = 9999
    while arg1 > 500 or b2>500:
        pwr1 = RandInt(2,5)
        pwr2 = NonZeroInt(2,5,[pwr1])
        while gcd(pwr1,pwr2)>1:
            pwr1 = RandInt(2,5)
            pwr2 = NonZeroInt(2,5,[pwr1])
        arg1 = b1^pwr1
        b2 = b1^pwr2
    ans1 = pwr1/pwr2
    return [b1,arg1,pwr1,pwr2,b2,ans1]
