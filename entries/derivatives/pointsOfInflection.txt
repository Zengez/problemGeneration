def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def pointsOfInflection():
    c1 = RandInt(-5,5)
    c2 = NonZeroInt(-5,5,[c1])
    c3 = NonZeroInt(-5,5,[c1,c2])
    lead = NonZeroInt(-5,5)
    f1der = expand(lead*(x-c1)*(x-c2)*(x-c3))
    fdisp = integral(integral(f1der,x) + RandInt(-10,10),x)+RandInt(-10,10)
    while ( abs(f1der.coefficient(x,1))>80 or abs(f1der.coefficient(x,2))>80 or abs(f1der.coefficient(x,3))>80 or abs(f1der.coefficient(x,4))>80 or abs(fdisp.coefficient(x,1))>80 or abs(fdisp.coefficient(x,2))>80 or abs(fdisp.coefficient(x,3))>80 or abs(fdisp.coefficient(x,4))>80 or abs(fdisp.coefficient(x,5))>80 ):
        c1 = RandInt(-5,5)
        c2 = NonZeroInt(-5,5,[c1])
        c3 = NonZeroInt(-5,5,[c1,c2])
        lead = NonZeroInt(-5,5)
        f1der = expand(lead*(x-c1)*(x-c2)*(x-c3))
        fdisp = integral(integral(f1der,x) + RandInt(-10,10),x)+RandInt(-10,10)
    ans = c1+c2+c3
    return [fdisp, ans]
