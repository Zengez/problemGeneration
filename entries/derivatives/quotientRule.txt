    """
    return QQ(randint(int(a),int(b)))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def quotientRule():
    var('x,y,z,X,Y,Z')
    f1 = RandInt(-10,10)*x^2 + RandInt(-10,10)*x^1 + RandInt(-10,10) + RandInt(-10,10)*x^(RandInt(-5,5)/NonZeroInt(-5,5))

    ans = derivative(f3,x)
