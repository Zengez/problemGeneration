CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def localExtrema():
    c1 = RandInt(-10,10)
    c2 = NonZeroInt(-10,10,[c1])
    c3 = NonZeroInt(-10,10,[c1,c2])
    flip = RandInt(0,1)
    f1der = expand((x-c1)*(x-c2)*(x-c3)*(-1)^flip*RandInt(1,4))
    f1disp = integral(f1der,x) + RandInt(-10,10)
    while (abs(f1disp.coefficient(x, 1))>80 or abs(f1disp.coefficient(x, 2))>80 or abs(f1disp.coefficient(x, 3))>80 or abs(f1disp.coefficient(x, 4))>80 or abs(f1disp(x=0))>80):
        c1 = RandInt(-10,10)
        c2 = NonZeroInt(-10,10,[c1])
        c3 = NonZeroInt(-10,10,[c1,c2])
        flip = RandInt(0,1)
        f1der = expand((x-c1)*(x-c2)*(x-c3)*(-1)^flip*RandInt(1,4))
        f1disp = integral(f1der,x) + RandInt(-10,10)
    ansMin = 2-flip
    ansMax = 1+flip
    sortedzeros = sorted([c1,c2,c3])
    if flip==1:
        ansMinxSum = sortedzeros[1]
        ansMaxxSum = sortedzeros[0]+sortedzeros[2]
        ansMinySum = f1disp(x=sortedzeros[1])
        ansMaxySum = f1disp(x=sortedzeros[0]) + f1disp(x=sortedzeros[2])
    if flip==0:
        ansMaxxSum = sortedzeros[1]
        ansMinxSum = sortedzeros[0]+sortedzeros[2]
        ansMaxySum = f1disp(x=sortedzeros[1])
        ansMinySum = f1disp(x=sortedzeros[0]) + f1disp(x=sortedzeros[2])
    return [f1disp, ansMin, ansMinxSum, ansMinySum, ansMax, ansMaxxSum, ansMaxySum]
