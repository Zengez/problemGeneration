def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def newtonianMechanics():
    var('t,x,y,z,X,Y,Z')
    f1 = NonZeroInt(-10,10)*t^3 + RandInt(-10,10)*t^2 + RandInt(-10,10)*t + RandInt(-10,10) + RandInt(-5,5)*e^(RandInt(-5,5)*t-RandInt(-5,5)) + RandInt(-5,5)*log(RandInt(-5,5)*t - RandInt(-5,5))
    ans1 = derivative(f1,t)
    ans2 = derivative(ans1,t)
    return [f1, ans1, ans2]

\begin{problem}

\begin{itemize}

\end{itemize}
\end{problem}
