def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def concavity():
    c1 = RandInt(-5,5)
    c2 = NonZeroInt(-5,5,[c1])
    c3 = NonZeroInt(-5,5,[c1,c2])
    derzeros = sorted([c1,c2,c3])
    lead = NonZeroInt(-5,5)
    f1der = expand(lead*(x-c1)*(x-c2)*(x-c3))
    fdisp = integral(integral(f1der,x) + RandInt(-10,10),x)+RandInt(-10,10)
    while ( abs(f1der.coefficient(x,1))>100 or abs(f1der.coefficient(x,2))>100 or abs(f1der.coefficient(x,3))>100 or abs(f1der.coefficient(x,4))>100 or abs(fdisp.coefficient(x,1))>100 or abs(fdisp.coefficient(x,2))>100 or abs(fdisp.coefficient(x,3))>100 or abs(fdisp.coefficient(x,4))>100 or abs(fdisp.coefficient(x,5))>100 ):
        c1 = RandInt(-5,5)
        c2 = NonZeroInt(-5,5,[c1])
        c3 = NonZeroInt(-5,5,[c1,c2])
        derzeros = sorted([c1,c2,c3])
        lead = NonZeroInt(-5,5)
        f1der = expand(lead*(x-c1)*(x-c2)*(x-c3))
        fdisp = integral(integral(f1der,x) + RandInt(-10,10),x)+RandInt(-10,10)
    if lead > 0:
        ansUpL1 = derzeros[0]
        ansUpR1 = derzeros[1]
        ansUpL2 = derzeros[2]
        ansUpR2 = infinity
        ansDownL1 = -infinity
        ansDownR1 = derzeros[0]
        ansDownL2 = derzeros[1]
        ansDownR2 = derzeros[2]
    if lead < 0:
        ansUpL1 = -infinity
        ansUpR1 = derzeros[0]
        ansUpL2 = derzeros[1]
        ansUpR2 = derzeros[2]
        ansDownL1 = derzeros[0]
        ansDownR1 = derzeros[1]
        ansDownL2 = derzeros[2]
        ansDownR2 = infinity
    return [fdisp, ansUpL1, ansUpR1, ansUpL2, ansUpR2, ansDownL1, ansDownR1, ansDownL2, ansDownR2]

\begin{problem}
Consider the function $\sage{p1fdisp}$. Over what intervals is the function:
\begin{itemize}
    \item Concave Up: $\left(\answer{\sage{p1ansUpL1}},\answer{\sage{p1ansUpR1}}\right) \cup \left(\answer{\sage{p1ansUpL2}},\answer{\sage{p1ansUpR2}}\right)$
    \item Concave Down: $\left(\answer{\sage{p1ansDownL1}},\answer{\sage{p1ansDownR1}}\right) \cup \left(\answer{\sage{p1ansDownL2}},\answer{\sage{p1ansDownR2}}\right)$
\end{itemize}
\end{problem}
