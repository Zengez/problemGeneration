def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def sigmaNotation():
    var('k,x,y,z,X,Y,Z')
    indexStart = RandInt(-4,4)
    indexEnd = RandInt(indexStart+1, 10)
    f1 = RandInt(-5,5)*x+RandInt(-5,5)*x^2+RandInt(-5,5)*e^x
    ans = sum(f1,x,indexStart,indexEnd)
    totalTerms = indexEnd-indexStart+1
    return [indexStart, indexEnd, f1, ans]
