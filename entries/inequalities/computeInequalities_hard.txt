def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def computeInequalitiesHard():
    var('x,y,z,X,Y,Z,r,R')
    var('f,g,h,dx,dy,dz,dh,df')
    c1 = NonZeroInt(-5,5)
    c2 = RandInt(-5,5)
    c3 = NonZeroInt(-5,5,[c1])
    c4 = RandInt(-5,5)

    fvec = [x, x^2, x^3, sqrt(x)]
    fp1 = RandInt(0,3)
    fp2 = NonZeroInt(0,3,[fp1])
    f1 = fvec[fp1]
    f2 = fvec[fp2]

    fLeft = NonZeroInt(-5,5)*f1(x=(c1*x-c2)) + RandInt(-5,5)
    fRight = NonZeroInt(-5,5)*f2(x=(c3*x-c4)) + RandInt(-5,5)

    # We want to make sure that, if a sqrt appears, it should be nice to evaluate.
    if fp1==3:
        target=RandInt(1,10)^2
        check = (target+c2)/c1
    elif fp2==3:
        target=RandInt(1,10)^2
        check = (target+c4)/c3
    else:
        check = RandInt(-10,10)


    while abs(fLeft(x=check))>300 or abs(fRight(x=check))>300:
        fp1 = RandInt(0,3)
        fp2 = NonZeroInt(0,3,[fp1])
        f1 = fvec[fp1]
        f2 = fvec[fp2]
        fLeft = NonZeroInt(-5,5)*f1(x=(c1*x-c2)) + RandInt(-5,5)
        fRight = NonZeroInt(-5,5)*f2(x=(c3*x-c4)) + RandInt(-5,5)
        if fp1==3:
            target=RandInt(1,10)^2
            check = (target+c2)/c1
        elif fp2==3:
            target=RandInt(1,10)^2
            check = (target+c4)/c3
        else:
            check = RandInt(-10,10)


    ## Determine which of the symbols we will use in the inequality from: [<, <=, >, >=]
    sym1Choice=RandInt(0,3)

    ## Assume it isn't true, then flip it to true if it is. Faster computation this way.
    ans=0
    if sym1Choice==0:
        fDisp = (fLeft < fRight)
        if fLeft(x=check)<fRight(x=check):
            ans=1
    if sym1Choice==1:
        fDisp = (fLeft <= fRight)
        if fLeft(x=check)<=fRight(x=check):
            ans=1
    if sym1Choice==2:
        fDisp = (fLeft > fRight)
        if fLeft(x=check)>fRight(x=check):
            ans=1
    if sym1Choice==3:
        fDisp = (fLeft >= fRight)
        if fLeft(x=check)>=fRight(x=check):
            ans=1
    return [check, fDisp, ans]
