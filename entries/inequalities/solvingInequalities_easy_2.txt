def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def solvingInequalitiesEasy():
    var('x,y,z,X,Y,Z,r,R')
    c1 = NonZeroInt(-5,5)
    c2 = RandInt(-5,5)
    c3 = NonZeroInt(-5,5)
    c4 = RandInt(-5,5)
    c5 = NonZeroInt(-5,5)
    c6 = RandInt(-5,5)
    fact1 = c1*x-c2
    fact2 = c3*x-c4
    fact3 = c5*x-c6
    f1 = expand(fact1*fact2*fact3)
    leftSplit = RandInt(0,1)
    if leftSplit==0:
        fLeft = f1.coefficient(x^3)*x^3 + f1(x=0)
        fRight = -f1.coefficient(x)*x - f1.coefficient(x^2)*x^2
    else:
        fLeft = f1.coefficient(x)*x + f1(x=0)
        fRight = -f1.coefficient(x^3)*x^3 - f1.coefficient(x^2)*x^2
    zeroVec = sorted([c2/c1,c4/c3,c6/c5])
    zero1=zeroVec[0]; zero2=zeroVec[1]; zero3=zeroVec[2]
    sym1Choice = RandInt(0,3)
    if sym1Choice==0: fDisp = (fLeft < fRight)
    if sym1Choice==1: fDisp = (fLeft > fRight)
    if sym1Choice==2: fDisp = (fLeft <= fRight)
    if sym1Choice==3: fDisp = (fLeft >= fRight)
    return [fDisp]
