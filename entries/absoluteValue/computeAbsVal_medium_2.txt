CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def computeAbsValMedium():
    ans = 500
    while abs(ans)>300:
        f1 = RandInt(-10,10)*x^2 + NonZeroInt(-5,5)*abs(RandInt(-5,5)*x^2 + NonZeroInt(-5,5)*x + RandInt(-10,10)) + NonZeroInt(-5,5)*x + RandInt(-10,10)
        target = RandInt(-5,5)
        ans = f1(x=target)
    return [f1, target, ans]
