def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def modelingDistanceHard():
    var('a','b','c','d')
    letterVec = [a,b,c,d]
    obj1 = NonZeroInt(-5,5)*choice(letterVec) + NonZeroInt(-5,5)*choice(letterVec) + NonZeroInt(-5,5)*choice(letterVec)
    obj2 = NonZeroInt(-5,5)*choice(letterVec) + NonZeroInt(-5,5)*choice(letterVec) + NonZeroInt(-5,5)*choice(letterVec)
    dist = RandInt(1,200)
    relateVec = [[LatexExpr(r'\text{at least}'),(abs(obj1 - obj2) >= dist)],[LatexExpr(r'\text{no less than}'),(abs(obj1 - obj2) >= dist)],[LatexExpr(r'\text{at most}'),(abs(obj1 - obj2) <= dist)],[LatexExpr(r'\text{no more than}'),(abs(obj1 - obj2) <= dist)],[LatexExpr(r'\text{strictly less than}'),(abs(obj1 - obj2) < dist)],[LatexExpr(r'\text{strictly more than}'),(abs(obj1 - obj2) > dist)]]
    relateChoice = RandInt(0,5)
    relateText=relateVec[relateChoice][0]
    ansExpr=relateVec[relateChoice][1]
    ansVec = [(abs(obj1 - obj2) > dist),(abs(obj1 - obj2) < dist),(abs(obj1 - obj2) >= dist),(abs(obj1 - obj2) <= dist),(abs(obj1 - dist) > obj2),(abs(obj1 - dist) < obj2),(abs(obj1 - dist) >= obj2),(abs(obj1 - dist) <= obj2),(abs(dist - obj2) < obj1),(abs(dist - obj2) > obj1),(abs(dist - obj2) <= obj1),(abs(dist - obj2) >= obj1)]
    shuffle(ansVec)
    ans = ansVec.index(ansExpr)+1
    return [obj1, obj2, relateText, dist, ansVec, ans]
