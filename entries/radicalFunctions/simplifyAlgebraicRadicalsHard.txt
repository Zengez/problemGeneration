TITLE: Simplifying Algebraic Radicals (hard)
TAGS: Radical Functions, Simplifying, hard

var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def simplifyAlgebraicRadicalsHard():
    rootVal = RandInt(2,9)
    c1=c2=c3=c4=c5=c6=c7=c8=2
    while abs(c2/c1)==abs(c4/c3) or abs(c2/c1)==abs(c6/c5) or abs(c2/c1)==abs(c8/c7) or abs(c4/c3)==abs(c6/c5) or abs(c4/c3)==abs(c8/c7) or abs(c6/c5)==abs(c8/c7):
        c1=c2=c3=c4=c5=c6=c7=c8=2
        while gcd(c1,c2)>1:
            c1 = NonZeroInt(1,5); c2 = RandInt(-10,10)
        while gcd(c3,c4)>1:
            c3 = NonZeroInt(1,5); c4 = RandInt(-10,10)
        while gcd(c5,c6)>1:
            c5 = NonZeroInt(1,5); c6 = RandInt(-10,10)
        while gcd(c7,c8)>1:
            c7 = NonZeroInt(1,5); c8 = RandInt(-10,10)
    f1remainPwr=RandInt(0,rootVal-1); f2remainPwr=RandInt(0,rootVal-1); f3remainPwr=RandInt(0,rootVal-1); f4remainPwr=RandInt(0,rootVal-1)
    f1factorPwr=RandInt(0,5); f2factorPwr=RandInt(0,5); f3factorPwr=RandInt(0,5); f4factorPwr=RandInt(0,5)
    f1remain=(c1*x-c2)^f1remainPwr; f2remain=(c3*x-c4)^f2remainPwr; f3remain=(c5*x-c6)^f3remainPwr; f4remain=(c7*x-c8)^f4remainPwr
    f1total=(c1*x-c2)^(f1remainPwr+f1factorPwr*rootVal)
    f2total=(c3*x-c4)^(f2remainPwr+f2factorPwr*rootVal)
    f3total=(c5*x-c6)^(f3remainPwr+f3factorPwr*rootVal)
    f4total=(c7*x-c8)^(f4remainPwr+f4factorPwr*rootVal)
    fDisp=f1total*f2total*f3total*f4total
    fRemainder=f1remain*f2remain*f3remain*f4remain
    if rootVal%2==0:
        f1factored=(c1*x-c2)^f1factorPwr if f1factorPwr%2==0 else abs(c1*x-c2)^f1factorPwr
        f2factored=(c3*x-c4)^f2factorPwr if f2factorPwr%2==0 else abs(c3*x-c4)^f2factorPwr
        f3factored=(c5*x-c6)^f3factorPwr if f3factorPwr%2==0 else abs(c5*x-c6)^f3factorPwr
        f4factored=(c7*x-c8)^f4factorPwr if f4factorPwr%2==0 else abs(c7*x-c8)^f4factorPwr
        fFactored=f1factored*f2factored*f3factored*f4factored
        hint=LatexExpr(r'\text{Since the root-value is even, you need absolute values around the term you simplified out. }')
        hint2=LatexExpr(r'\text{Then you can remove those absolute values if the power of the outside factor is even.}')
    else:
        fFactored=(c1*x-c2)^f1factorPwr*(c3*x-c4)^f2factorPwr*(c5*x-c6)^f3factorPwr*(c7*x-c8)^f4factorPwr
        hint=LatexExpr(r'\text{Since the root-value is odd, you are all set.}')
        hint2=LatexExpr(r' ')
    return [rootVal,fDisp,fFactored,fRemainder,hint,hint2]

\begin{problem}
    \begin{hint}
        Group up the factor(s) in the radicand into the largest multiple of the root-value (that is at most the original power of the term in the radicand) and whatever remainder is left over after you separate out that many of the factor.
    \end{hint}
    \begin{hint}
        Pull the portion of the factor that is a multiple of the root-value outside of the radical, dividing the power by the root-value (the result should be a whole number, since it is a multiple of the root-value).
    \end{hint}

        $\sage{p1hint2}$
    \end{hint}
    Simplify the following radical. Notice that the root symbol is already supplied for you so you only need to supply the inside and outside functions (no need to expand them!)
    \[
        \sqrt[\sage{p1rootVal}]{\sage{p1fDisp}} = \left(\answer{\sage{p1fFactored}}\right)\sqrt[\sage{p1rootVal}]{\answer{\sage{p1fRemainder}}}
    \]
\end{problem}
