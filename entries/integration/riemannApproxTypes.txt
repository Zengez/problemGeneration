def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def riemannApproxTypes():
    var('k,x,y,z,X,Y,Z')
    funcvec = [x,x^2,x^3,e^x]
    ansREP = 111111/9999998
    ansLEP = 0
    ansWP = 0
    while abs(numerator(ansREP))>1000 or abs(denominator(ansREP))>1000 or abs(numerator(ansLEP))>1000 or abs(denominator(ansLEP))>1000 or abs(numerator(ansWP))>1000 or abs(denominator(ansWP))> 1000:
        sel = RandInt(0,3)
        RecNum = RandInt(3,6)
        start = RandInt(-5,5)
        width = RandInt(1,3)/2
        end = start+width*RecNum
        f1 = NonZeroInt(-5,5)* funcvec[sel](x=(x - (start+width+RandInt(-3,3)/2)) ) + RandInt(-5,5)
        ansREP = sum(width*f1(x=(start+k*width)),k,1,RecNum)
        ansLEP = sum(width*f1(x=(start+k*width)),k,0,RecNum-1)
        ansMP = sum(width*f1(x=(start+(k-1/2)*width)),k,1,RecNum)
    return [RecNum, f1, start, end, ansREP, ansLEP, ansMP]
