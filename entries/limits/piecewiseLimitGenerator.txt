TITLE: Piecewise One-Sided Limits
TAGS: Limits, Piecewise Functions, One-Sided Limits

EXAMPLE:

CODE-SAGE:
var('x')

def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def piecewiseLimitGenerator():
    funcvec = [x, x^2, x^3, e^x, 2^x, log(abs(x)), sqrt(abs(x)), abs(x)]
    pick1 = RandInt(0,7)
    pick2 = RandInt(0,7)
    pick3 = RandInt(0,7)
    c1 = RandInt(-5,5)
    c2 = RandInt(-5,5)
    c3 = RandInt(-5,5)
    c4 = RandInt(-5,5)
    c5 = RandInt(-5,5)
    c6 = RandInt(-5,5)
    f1t = funcvec[pick1]
    f2t = funcvec[pick2]
    f3t = funcvec[pick3]
    f1t2 = f1t(x=(x-c1))+c2
    f2 = f2t(x=(x-c3))+c4
    f3t2 = f3t(x=(x-c5))+c6
    pivot1 = NonZeroInt(-5,1)
    pivot2 = NonZeroInt(pivot1+1, 5)
    flip = RandInt(0,1)
    flip2 = RandInt(0,1)
    temp1 = f2(x=(pivot1))-f1t2(x=(pivot1))
    temp2 = f2(x=(pivot2))-f3t2(x=(pivot2))
    f1 = f1t2 + temp1*flip
    f3 = f3t2 + temp2*flip2
    anspvt1L = f1(x=(pivot1))
    anspvt1R = f2(x=(pivot1))
    anspvt2L = f2(x=(pivot2))
    anspvt2R = f3(x=(pivot2))
    if anspvt1L==anspvt1R:
        anslim1 = anspvt1L
    else:
        anslim1 = LatexExpr(r"DNE")
    if anspvt2L==anspvt2R:
        anslim2 = anspvt2L
    else:
        anslim2 = LatexExpr(r"DNE")
    return [f1,f2,f3,pivot1,pivot2,anspvt1L,anspvt1R,anspvt2L,anspvt2R,anslim1,anslim2]

For the next several problems, consider the following function
\[
    g(x) =
    \begin{cases}
        \sage{p1f1} & x<\sage{p1pivot1}\\
        \sage{p1f2} &\sage{p1pivot1} \leq x < \sage{p1pivot2}\\
        \sage{p1f3} &\sage{p1pivot2}<x\\
    \end{cases}
\]
Determine if the following limits exist. If they do, compute their values.
\begin{problem}
    Does $\lim\limits_{x\rightarrow\sage{p1pivot1}^-} g(x)$ exist?
    \begin{multipleChoice}
        \choice[correct]{Yes.}
        \choice{No.}
    \end{multipleChoice}
    \begin{problem}
        $\lim\limits_{x\rightarrow\sage{p1pivot1}^-} g(x) = \answer{\sage{p1anspvt1L}}$
    \end{problem}
\end{problem}

\begin{problem}
    Does $\lim\limits_{x\rightarrow\sage{p1pivot1}^+} g(x)$ exist?
    \begin{multipleChoice}
        \choice[correct]{Yes.}
        \choice{No.}
    \end{multipleChoice}
    \begin{problem}
        $\lim\limits_{x\rightarrow\sage{p1pivot1}^+} g(x) = \answer{\sage{p1anspvt1R}}$
    \end{problem}
\end{problem}

\begin{problem}
    Does $\lim\limits_{x\rightarrow\sage{p1pivot2}^-} g(x)$ exist?
    \begin{multipleChoice}
        \choice[correct]{Yes.}
        \choice{No.}
    \end{multipleChoice}
    \begin{problem}
        $\lim\limits_{x\rightarrow\sage{p1pivot2}^-} g(x) = \answer{\sage{p1anspvt2L}}$
    \end{problem}
\end{problem}

\begin{problem}
    Does $\lim\limits_{x\rightarrow\sage{p1pivot2}^+} g(x)$ exist?
    \begin{multipleChoice}
        \choice[correct]{Yes.}
        \choice{No.}
    \end{multipleChoice}
    \begin{problem}
        $\lim\limits_{x\rightarrow\sage{p1pivot2}^+} g(x) = \answer{\sage{p1anspvt2R}}$
    \end{problem}
\end{problem}

\begin{problem}
    Compute $\lim\limits_{x\rightarrow\sage{p1pivot1}} g(x)$. If it does not exist, write 'DNE'
    $\lim\limits_{x\rightarrow\sage{p1pivot1}} g(x) = \answer{\sage{p1anslim1}}$.
\end{problem}
