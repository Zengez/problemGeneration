def RandInt(a,b):
    """ Returns a random integer in [`a`,`b`]. Note that `a` and `b` should be integers themselves to avoid unexpected behavior.
    """
    return QQ(randint(int(a),int(b)))
    # return choice(range(a,b+1))

def NonZeroInt(b,c, avoid = [0]):
    """ Returns a random integer in [`b`,`c`] which is not in `av`. 
        If `av` is not specified, defaults to a non-zero integer.
    """
    while True:
        a = RandInt(b,c)
        if a not in avoid:
            return a

def hAOneEasy():
    var('x,y,z,X,Y,Z')
    target = RandInt(-10,10)
    c1 = NonZeroInt(1,10,[0,target])
    c2 = NonZeroInt(1,10,[0,target])
    c3 = NonZeroInt(1,10,[target])
    c4 = NonZeroInt(1,10,[target])
    c5 = NonZeroInt(-30,30)
    c6 = NonZeroInt(-30,30)
    n1 = RandInt(1,5)
    n2 = RandInt(1,5)
    f1top = expand(c5*(x^n1-c1)*(x-c2)*(x-target))
    f1bot = expand(c6*(x^n2-c3)*(x-c4)*(x-target))
    if n1 > n2:
        leftHA = LatexExpr(r"DNE"); rightHA = LatexExpr(r"DNE")
    if n1 < n2:
        leftHA = 0; rightHA = 0
    if n1==n2:
        leftHA = c5/c6; rightHA = c5/c6
    return [f1top, f1bot, leftHA, rightHA]
